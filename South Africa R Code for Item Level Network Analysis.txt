# Item-Level Symptom Network Analysis (South Africa Sample)
# USING CPSS data
# GGM with EBICglasso regularisation, packages: bootnet, qgraph

# 1. Installing & loading packages

packages <- c("bootnet", "qgraph", "dplyr", "tidyr")

install_if_missing <- function(pkg) {
  if (!requireNamespace(pkg, quietly = TRUE)) {
    install.packages(pkg, dependencies = TRUE, repos = "https://cloud.r-project.org")
  }
}

invisible(lapply(packages, install_if_missing))

library(bootnet)
library(qgraph)
library(dplyr)


# 2. Loading data
# Setting path to CSV file

data_path <- "Data_South_Africa_CTSR_GAD_PHQ_CPSS.csv"

df <- read.csv(data_path, header = TRUE, stringsAsFactors = FALSE)

cat("Data dimensions:", nrow(df), "rows x", ncol(df), "columns\n")
cat("Variables:", paste(names(df), collapse = ", "), "\n\n")


# 3. Selecting and ordering variables
# CTS (orange), GAD (green), PHQ (pink), CPSS (blue)


cts_vars  <- grep("^A_CTSR_", names(df), value = TRUE)
gad_vars  <- grep("^A_GAD_", names(df), value = TRUE)
phq_vars  <- grep("^A_PHQ_", names(df), value = TRUE)
cpss_vars <- grep("^A_CPSS_", names(df), value = TRUE)

cat("CTS items :", length(cts_vars), "\n")
cat("GAD items :", length(gad_vars), "\n")
cat("PHQ items :", length(phq_vars), "\n")
cat("CPSS items:", length(cpss_vars), "\n\n")

# Order: GAD | CTS | PHQ | CPSS
node_order <- c(gad_vars, cts_vars, phq_vars, cpss_vars)

# Select data
data_net <- df[, node_order]

# Remove rows with all NA
data_net <- data_net[rowSums(is.na(data_net)) < ncol(data_net), ]

cat("Rows after removing all-NA rows:", nrow(data_net), "\n\n")


# 4. Defining node groups and colours

n_gad  <- length(gad_vars)
n_cts  <- length(cts_vars)
n_phq  <- length(phq_vars)
n_cpss <- length(cpss_vars)

groups <- c(
  rep("Anxiety (GAD)", n_gad),
  rep("CTS", n_cts),
  rep("Depression (PHQ)", n_phq),
  rep("PTSD (CPSS)", n_cpss)
)

group_colours <- c(
  "Anxiety (GAD)"    = "#66C266",
  # Green
  "CTS"              = "#F4A0B0",
  # Pink
  "Depression (PHQ)" = "#6AAFE6",
  # Blue
  "PTSD (CPSS)"      = "#F4A929"    
  # Orange
)

node_colours <- group_colours[groups]


# 5. Creating node labels (remove "A_" prefix from dataset)


make_label <- function(varname) {
  # Remove leading "A_" only
  sub("^A_", "", varname)
}

node_labels <- sapply(node_order, make_label)


# 6. Estimating GGM with EBICglasso via bootnet

cat("Estimating GGM network (EBICglasso) ...\n")

set.seed(42)
network <- estimateNetwork(
  data        = data_net,
  default     = "EBICglasso",
  corMethod   = "cor",
  # Pearson; use "spearman" for ordinal if preferred
  tuning      = 0.5,
  # EBIC hyperparameter (γ = 0.5 is standard)
  missing     = "pairwise",
  verbose     = TRUE
)

cat("\nNetwork estimated successfully.\n")
cat("Number of non-zero edges:", sum(network$graph[upper.tri(network$graph)] != 0), "\n\n")

# 8. Network descriptive statistics

cat("=== Network Descriptive Statistics ===\n\n")

# Edge weight matrix
W <- network$graph

# Strength (sum of absolute edge weights for each node)
strength <- colSums(abs(W))
cat("Node Strength (top 10):\n")
print(round(sort(strength, decreasing = TRUE)[1:10], 3))

# Density
n_nodes   <- ncol(W)
max_edges <- n_nodes * (n_nodes - 1) / 2
n_edges   <- sum(W[upper.tri(W)] != 0)
density   <- n_edges / max_edges
cat(sprintf(
  "\nDensity: %.3f (%d / %d possible edges)\n",
  density,
  n_edges,
  max_edges
))

# Edge weight summary
edge_weights <- W[upper.tri(W)]
nonzero_edges <- edge_weights[edge_weights != 0]
cat(sprintf("\nNon-zero edge weights (n=%d):\n", length(nonzero_edges)))
print(summary(nonzero_edges))


# FULL CTS CROSS-DOMAIN EDGE EXTRACTION


# Defining domain indices
ptsd_indices <- which(groups == "PTSD (CPSS)")
phq_indices  <- which(groups == "Depression (PHQ)")
gad_indices  <- which(groups == "Anxiety (GAD)")


# Function to extract ALL non-zero edges between two sets

extract_edges <- function(mat, rows, cols, row_labels, col_labels, domain_name) {
  
  submat <- mat[rows, cols]
  
  edge_list <- which(submat != 0, arr.ind = TRUE)
  
  if (nrow(edge_list) == 0) {
    return(NULL)
  }
  
  data.frame(
    CTS_Item   = row_labels[rows][edge_list[,1]],
    Connected_To = col_labels[cols][edge_list[,2]],
    Weight     = submat[edge_list],
    Abs_Weight = abs(submat[edge_list]),
    Domain     = domain_name
  )
}


# Extract all bridges

cts_ptsd_edges <- extract_edges(
  W,
  cts_indices,
  ptsd_indices,
  node_labels,
  node_labels,
  "PTSD"
)

cts_phq_edges <- extract_edges(
  W,
  cts_indices,
  phq_indices,
  node_labels,
  node_labels,
  "Depression"
)

cts_gad_edges <- extract_edges(
  W,
  cts_indices,
  gad_indices,
  node_labels,
  node_labels,
  "Anxiety"
)

# Combine into one dataframe
all_cts_bridges <- rbind(cts_ptsd_edges,
                         cts_phq_edges,
                         cts_gad_edges)

# Sort by absolute weight, strongest first
all_cts_bridges <- all_cts_bridges[order(-all_cts_bridges$Abs_Weight), ]

# Print results
cat("\n====================================================\n")
cat("ALL CTS CROSS-DOMAIN BRIDGES (sorted by strength)\n")
cat("====================================================\n\n")

print(all_cts_bridges)

# Save to CSV
write.csv(all_cts_bridges,
          "South_Africa_All_CTS_CrossDomain_Edges.csv",
          row.names = FALSE)

cat("\nSaved full bridge table to: South_Africa_All_CTS_CrossDomain_Edges.csv\n")

# START OF BRIDGING CODE:

# Computing maximum cross-domain edge for each CTS item

cts_bridge <- sapply(cts_indices, function(i) {
  max(abs(W[i, non_cts_indices]))
})
# Create dataframe
bridge_df <- data.frame(CTS_Item = node_labels[cts_indices], Max_Cross_Domain_Edge = cts_bridge)
# Sort descending
bridge_df <- bridge_df[order(-bridge_df$Max_Cross_Domain_Edge), ]
print(bridge_df)

# Identify top 3 only
bridge_df$Category <- "Other"

# Reverse order for horizontal plot
