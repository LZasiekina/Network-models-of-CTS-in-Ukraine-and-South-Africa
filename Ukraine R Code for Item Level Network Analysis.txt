  missing     = "pairwise",
  verbose     = TRUE
)

# FULL CTS CROSS-DOMAIN EDGE EXTRACTION


# Defining domain indices
ptsd_indices <- which(groups == "PTSD (CPSS)")
phq_indices  <- which(groups == "Depression (PHQ)")
gad_indices  <- which(groups == "Anxiety (GAD)")


# Extracting ALL non-zero edges between two sets

extract_edges <- function(mat, rows, cols, row_labels, col_labels, domain_name) {
  
  submat <- mat[rows, cols]
  
  edge_list <- which(submat != 0, arr.ind = TRUE)
  
  if (nrow(edge_list) == 0) {
    return(NULL)
  }
  
  data.frame(
    CTS_Item   = row_labels[rows][edge_list[,1]],
    Connected_To = col_labels[cols][edge_list[,2]],
    Weight     = submat[edge_list],
    Abs_Weight = abs(submat[edge_list]),
    Domain     = domain_name
  )
}


# Extracting all bridges

cts_ptsd_edges <- extract_edges(
  W,
  cts_indices,
  ptsd_indices,
  node_labels,
  node_labels,
  "PTSD"
)

cts_phq_edges <- extract_edges(
  W,
  cts_indices,
  phq_indices,
  node_labels,
  node_labels,
  "Depression"
)

cts_gad_edges <- extract_edges(
  W,
  cts_indices,
  gad_indices,
  node_labels,
  node_labels,
  "Anxiety"
)

# Combining into one dataframe
all_cts_bridges <- rbind(cts_ptsd_edges,
                         cts_phq_edges,
                         cts_gad_edges)

# Sorting by absolute weight (strongest first)
all_cts_bridges <- all_cts_bridges[order(-all_cts_bridges$Abs_Weight), ]
